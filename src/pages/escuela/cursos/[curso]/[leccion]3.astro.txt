---
export const prerender = false;
import { env } from 'cloudflare:workers';
import Layout from '../../../../layouts/Layout.astro';
import { accountsEnabled, getCustomerEmail } from '../../../../features/auth/customer';
import { renderMarkdown } from '../../../../features/pages/markdown';

const courseSlug = Astro.params.curso ?? '';
const lessonSlug = Astro.params.leccion ?? '';
if (!(await accountsEnabled())) return Astro.redirect('/', 303);
const email = await getCustomerEmail(Astro.cookies);
if (!email) return Astro.redirect('/account/login', 303);
const subscription = await env.DB.prepare(`SELECT status FROM school_subscriptions WHERE student_email = ? COLLATE NOCASE AND status IN ('active','trialing') AND (current_period_end IS NULL OR current_period_end > strftime('%Y-%m-%dT%H:%M:%fZ','now')) LIMIT 1`).bind(email).first();
if (!subscription) return Astro.redirect('/escuela/planes', 303);
const row = await env.DB.prepare(`SELECT c.id AS course_id, c.title AS course_title, c.slug AS course_slug, l.id AS lesson_id, l.title AS lesson_title, l.summary, l.content_md, l.content_html, l.sequence_no FROM school_courses c JOIN school_lessons l ON l.course_id = c.id WHERE c.slug = ? AND l.slug = ? AND c.status = 'published' AND l.status = 'published' LIMIT 1`).bind(courseSlug, lessonSlug).first<{course_id:string;course_title:string;course_slug:string;lesson_id:string;lesson_title:string;summary:string;content_md:string;content_html:string;sequence_no:number}>();
if (!row) return Astro.redirect('/escuela/mi-escuela', 303);
// content_html is generated on the server with markdown-it html:false. Older
// lessons without the new column value fall back to the same safe renderer.
const safeHtml = row.content_html || renderMarkdown(row.content_md || '');
const next = await env.DB.prepare(`SELECT slug, title FROM school_lessons WHERE course_id = ? AND status = 'published' AND sequence_no > ? ORDER BY sequence_no LIMIT 1`).bind(row.course_id, row.sequence_no).first<{slug:string;title:string}>();
const previous = await env.DB.prepare(`SELECT slug, title FROM school_lessons WHERE course_id = ? AND status = 'published' AND sequence_no < ? ORDER BY sequence_no DESC LIMIT 1`).bind(row.course_id, row.sequence_no).first<{slug:string;title:string}>();
---
<Layout title={row.lesson_title} description={row.summary || row.course_title}>
  <main class="mx-auto w-full max-w-4xl px-4 py-8 sm:px-6 lg:py-12">
    <a class="text-sm text-kumo-muted hover:underline" href={`/escuela/mi-escuela?curso=${encodeURIComponent(row.course_slug)}`}>← Volver al curso</a>
    <header class="mb-8 mt-5 border-b border-kumo-line pb-6">
      <p class="text-sm text-kumo-muted">{row.course_title} · Lección {row.sequence_no}</p>
      <h1 class="mt-2 text-3xl font-bold tracking-tight">{row.lesson_title}</h1>
      {row.summary && <p class="mt-3 text-kumo-muted">{row.summary}</p>}
    </header>
    <article class="school-lesson-content max-w-none" set:html={safeHtml}></article>
    <nav class="mt-10 flex flex-wrap items-center justify-between gap-4 border-t border-kumo-line pt-6">
      {previous ? <a class="rounded-lg border border-kumo-line px-4 py-2 text-sm" href={`/escuela/cursos/${row.course_slug}/${previous.slug}`}>← {previous.title}</a> : <span></span>}
      {next ? <a class="rounded-lg bg-kumo-inverted px-4 py-2 text-sm text-kumo-inverted-foreground" href={`/escuela/cursos/${row.course_slug}/${next.slug}`}>{next.title} →</a> : <a class="rounded-lg border border-kumo-line px-4 py-2 text-sm" href={`/escuela/mi-escuela?curso=${encodeURIComponent(row.course_slug)}`}>Terminar lección</a>}
    </nav>
  </main>

    .school-lesson-content h2 { font-size: 1.4rem; }

    .school-lesson-content ul,.school-lesson-content ol { margin: 1rem 0; padding-left: 1.5rem; list-style: revert; }

    .school-lesson-content pre { overflow-x: auto; padding: 1rem; background: #f5f5f5; border-radius: .5rem; }
    .school-lesson-content img { max-width: 100%; height: auto; }
    .school-lesson-content table { border-collapse: collapse; max-width: 100%; }
