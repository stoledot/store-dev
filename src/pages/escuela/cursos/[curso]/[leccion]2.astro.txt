---
export const prerender = false;

import { env } from 'cloudflare:workers';
import MarkdownIt from 'markdown-it';
import Layout from '../../../../layouts/Layout.astro';
import { accountsEnabled, getCustomerEmail } from '../../../../features/auth/customer';

interface CourseRow {
  id: string;
  slug: string;
  title: string;
  summary: string;
}

interface LessonRow {
  id: string;
  course_id: string;
  slug: string;
  title: string;
  summary: string;
  content_md: string;
  sequence_no: number;
  lesson_type: string;
  required_to_complete: number;
}

interface NeighborRow {
  slug: string;
  title: string;
}

const courseSlug = Astro.params.curso ?? '';
const lessonSlug = Astro.params.leccion ?? '';

if (!(await accountsEnabled())) return Astro.redirect('/account/login', 303);
const email = await getCustomerEmail(Astro.cookies);
if (!email) {
  const next = `${Astro.url.pathname}${Astro.url.search}`;
  return Astro.redirect(`/account/login?next=${encodeURIComponent(next)}`, 303);
}

// El acceso se comprueba en el servidor en cada petición. El correo solo procede
// de la sesión firmada de MinShop, nunca de parámetros del navegador.
const subscription = await env.DB.prepare(`
  SELECT 1 AS active
    FROM school_subscriptions
   WHERE student_email = ? COLLATE NOCASE
     AND status IN ('active', 'trialing')
     AND (current_period_end IS NULL OR current_period_end > strftime('%Y-%m-%dT%H:%M:%fZ', 'now'))
   LIMIT 1
`).bind(email).first<{ active: number }>();

if (!subscription) return Astro.redirect('/escuela/mi-escuela/', 303);

const course = await env.DB.prepare(`
  SELECT id, slug, title, summary
    FROM school_courses
   WHERE slug = ? AND status = 'published'
   LIMIT 1
`).bind(courseSlug).first<CourseRow>();

if (!course) return Astro.redirect('/escuela/mi-escuela/', 303);

const lesson = await env.DB.prepare(`
  SELECT id, course_id, slug, title, summary, content_md, sequence_no,
         lesson_type, required_to_complete
    FROM school_lessons
   WHERE course_id = ? AND slug = ? AND status = 'published'
   LIMIT 1
`).bind(course.id, lessonSlug).first<LessonRow>();

if (!lesson) return Astro.redirect('/escuela/mi-escuela/', 303);

// Acepta la finalización solo mediante POST del mismo origen.
if (Astro.request.method === 'POST') {
  const origin = Astro.request.headers.get('origin');
  if (!origin || new URL(origin).origin !== Astro.url.origin) {
    return new Response('Solicitud no válida.', { status: 403 });
  }

  const now = new Date().toISOString();
  const id = crypto.randomUUID();
  await env.DB.batch([
    env.DB.prepare(`
      INSERT INTO school_lesson_progress
        (id, student_email, lesson_id, status, started_at, completed_at,
         content_version, last_activity_at, created_at, updated_at)
      VALUES (?, ? COLLATE NOCASE, ?, 'completed', ?, ?, 1, ?, ?, ?)
      ON CONFLICT(student_email, lesson_id) DO UPDATE SET
        status = 'completed',
        completed_at = COALESCE(school_lesson_progress.completed_at, excluded.completed_at),
        last_activity_at = excluded.last_activity_at,
        updated_at = excluded.updated_at
    `).bind(id, email, lesson.id, now, now, now, now, now),
    env.DB.prepare(`
      INSERT INTO school_course_progress
        (id, student_email, course_id, status, started_at, content_version, created_at, updated_at)
      VALUES (?, ? COLLATE NOCASE, ?, 'in_progress', ?, 1, ?, ?)
      ON CONFLICT(student_email, course_id) DO UPDATE SET
        status = CASE WHEN school_course_progress.status = 'completed' THEN 'completed' ELSE 'in_progress' END,
        started_at = COALESCE(school_course_progress.started_at, excluded.started_at),
        updated_at = excluded.updated_at
    `).bind(crypto.randomUUID(), email, course.id, now, now, now),
  ]);

  // Si ya se completaron todas las lecciones obligatorias, marcar el curso completo.
  const remaining = await env.DB.prepare(`
    SELECT COUNT(*) AS n
      FROM school_lessons l
      LEFT JOIN school_lesson_progress lp
        ON lp.lesson_id = l.id AND lp.student_email = ? COLLATE NOCASE
     WHERE l.course_id = ? AND l.status = 'published'
       AND l.required_to_complete = 1
       AND COALESCE(lp.status, 'not_started') <> 'completed'
  `).bind(email, course.id).first<{ n: number }>();

  if ((remaining?.n ?? 0) === 0) {
    await env.DB.prepare(`
      UPDATE school_course_progress
         SET status = 'completed', completed_at = COALESCE(completed_at, ?), updated_at = ?
       WHERE student_email = ? COLLATE NOCASE AND course_id = ?
    `).bind(now, now, email, course.id).run();
  }

  return Astro.redirect(Astro.url.pathname, 303);
}

const neighbors = await env.DB.prepare(`
  SELECT slug, title, sequence_no
    FROM school_lessons
   WHERE course_id = ? AND status = 'published'
   ORDER BY sequence_no ASC
`).bind(course.id).all<NeighborRow & { sequence_no: number }>();
const orderedLessons = neighbors.results ?? [];
const index = orderedLessons.findIndex((item) => item.slug === lesson.slug);
const previous = index > 0 ? orderedLessons[index - 1] : null;
const next = index >= 0 && index < orderedLessons.length - 1 ? orderedLessons[index + 1] : null;

const progress = await env.DB.prepare(`
  SELECT status FROM school_lesson_progress
   WHERE student_email = ? COLLATE NOCASE AND lesson_id = ?
   LIMIT 1
`).bind(email, lesson.id).first<{ status: string }>();

const md = new MarkdownIt({ html: false, linkify: false, breaks: false });
const lessonHtml = md.render(lesson.content_md || 'Esta lección todavía no tiene contenido.');
---

<Layout title={`${lesson.title} · ${course.title}`} description={lesson.summary || course.summary}>
  <main class="mx-auto w-full max-w-4xl px-4 py-8 sm:px-6 lg:py-12">
    <nav class="mb-6 text-sm text-kumo-muted" aria-label="Ruta de navegación">
      <a class="underline underline-offset-4" href="/escuela/mi-escuela/">Mi escuela</a>
      <span class="px-2">/</span>
      <a class="underline underline-offset-4" href={`/escuela/mi-escuela/?curso=${encodeURIComponent(course.slug)}`}>{course.title}</a>
    </nav>

    <header class="mb-8">
      <p class="text-sm font-medium text-kumo-muted">Lección {lesson.sequence_no} · {course.title}</p>
      <h1 class="mt-2 text-3xl font-bold tracking-tight sm:text-4xl">{lesson.title}</h1>
      {lesson.summary && <p class="mt-3 text-base leading-7 text-kumo-muted">{lesson.summary}</p>}
      <p class="mt-3 text-xs text-kumo-muted">{progress?.status === 'completed' ? 'Lección completada' : 'Lección en curso'}</p>
    </header>

    <article class="rounded-xl border border-kumo-line bg-kumo-base p-5 sm:p-8">
      <div class="course-content" set:html={lessonHtml}></div>
    </article>

    <div class="mt-8 flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
      {previous ? (
        <a class="inline-flex justify-center rounded-lg border border-kumo-line px-4 py-3 text-sm font-medium" href={`/escuela/cursos/${encodeURIComponent(course.slug)}/${encodeURIComponent(previous.slug)}`}>
          ← Lección anterior
        </a>
      ) : <span />}

      {lesson.required_to_complete === 1 && progress?.status !== 'completed' ? (
        <form method="POST">
          <button class="w-full rounded-lg bg-kumo-inverted px-5 py-3 text-sm font-semibold text-kumo-inverted-foreground sm:w-auto" type="submit">
            Marcar como completada
          </button>
        </form>
      ) : next ? (
        <a class="inline-flex justify-center rounded-lg bg-kumo-inverted px-5 py-3 text-sm font-semibold text-kumo-inverted-foreground" href={`/escuela/cursos/${encodeURIComponent(course.slug)}/${encodeURIComponent(next.slug)}`}>
          Siguiente lección →
        </a>
      ) : (
        <a class="inline-flex justify-center rounded-lg border border-kumo-line px-4 py-3 text-sm font-medium" href="/escuela/mi-escuela/">
          Volver a Mi escuela
        </a>
      )}
    </div>
  </main>
  <style>
    .course-content :global(h1), .course-content :global(h2), .course-content :global(h3) { margin: 1.5rem 0 .75rem; font-weight: 700; line-height: 1.3; }
    .course-content :global(h1) { font-size: 1.75rem; }
    .course-content :global(h2) { font-size: 1.4rem; }
    .course-content :global(h3) { font-size: 1.15rem; }
    .course-content :global(p), .course-content :global(ul), .course-content :global(ol), .course-content :global(blockquote) { margin: .9rem 0; line-height: 1.8; }
    .course-content :global(ul), .course-content :global(ol) { padding-left: 1.5rem; }
    .course-content :global(ul) { list-style: disc; }
    .course-content :global(ol) { list-style: decimal; }
    .course-content :global(a) { text-decoration: underline; text-underline-offset: 3px; }
    .course-content :global(pre) { overflow-x: auto; border: 1px solid var(--color-kumo-line); border-radius: .5rem; padding: 1rem; }
    .course-content :global(code) { font-family: ui-monospace, monospace; }
    .course-content :global(blockquote) { border-left: 3px solid var(--color-kumo-line); padding-left: 1rem; color: var(--color-kumo-muted); }
  </style>
</Layout>
