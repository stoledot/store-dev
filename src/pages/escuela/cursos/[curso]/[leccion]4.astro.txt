---
export const prerender = false;
import { env } from 'cloudflare:workers';
import Layout from '../../../../layouts/Layout.astro';
import { accountsEnabled, getCustomerEmail } from '../../../../features/auth/customer';
import { renderMarkdown } from '../../../../features/pages/markdown';
if (!(await accountsEnabled())) return Astro.redirect('/', 303);
const email = await getCustomerEmail(Astro.cookies);
if (!email) return Astro.redirect('/account/login', 303);
const courseSlug = Astro.params.curso ?? '';
const lessonSlug = Astro.params.leccion ?? '';
const access = await env.DB.prepare(`SELECT c.id AS course_id,c.title AS course_title,c.slug AS course_slug,c.level_code,a.title AS area_title FROM school_course_assignments x JOIN school_courses c ON c.id=x.course_id JOIN school_student_intake i ON i.student_email=x.student_email COLLATE NOCASE AND i.status='approved' AND i.selected_area_id=c.area_id AND i.approved_level=c.level_code LEFT JOIN school_areas a ON a.id=c.area_id WHERE x.student_email=? COLLATE NOCASE AND x.status='active' AND c.slug=? AND c.status='published' AND EXISTS (SELECT 1 FROM school_fee_payments p WHERE p.student_email=x.student_email COLLATE NOCASE AND p.fee_type='enrollment' AND p.valid_until>=date('now')) AND EXISTS (SELECT 1 FROM school_fee_payments p WHERE p.student_email=x.student_email COLLATE NOCASE AND p.fee_type='monthly' AND p.valid_until>=date('now')) LIMIT 1`).bind(email, courseSlug).first<{course_id:string;course_title:string;course_slug:string;level_code:string;area_title:string|null}>();
if (!access) return Astro.redirect('/escuela/mi-escuela', 303);
const lesson = await env.DB.prepare(`SELECT id,slug,title,summary,content_md,content_html,sequence_no FROM school_lessons WHERE course_id=? AND slug=? AND status='published'`).bind(access.course_id, lessonSlug).first<{id:string;slug:string;title:string;summary:string;content_md:string;content_html:string;sequence_no:number}>();
if (!lesson) return Astro.redirect('/escuela/mi-escuela', 303);
const lessonHtml = lesson.content_html || renderMarkdown(lesson.content_md);
const nav = (await env.DB.prepare(`SELECT slug,title,sequence_no FROM school_lessons WHERE course_id=? AND status='published' ORDER BY sequence_no`).bind(access.course_id).all<{slug:string;title:string;sequence_no:number}>()).results ?? [];
const index = nav.findIndex(l=>l.slug===lessonSlug);
const previous = index>0 ? nav[index-1] : null;
const next = index>=0 && index<nav.length-1 ? nav[index+1] : null;
---
<Layout title={lesson.title} description={lesson.summary || lesson.title}>
  <main class="mx-auto max-w-4xl px-4 py-8 sm:px-6 lg:py-12">
      <p class="text-xs uppercase tracking-widest text-kumo-muted">
        <a href="/escuela/mi-escuela" class="underline">Mi escuela</a> 
        / {access.area_title} / {access.level_code}</p>
      <h1 class="mt-3 text-3xl font-bold">{lesson.title}</h1>
      {lesson.summary && <p class="mt-3 text-kumo-muted">{lesson.summary}</p>}
      
      <article id="lesson-content" class="prose mt-8 max-w-none" set:html={lessonHtml}>
      </article>
        
      <nav class="mt-10 flex flex-wrap justify-between gap-4 border-t border-kumo-line pt-5">
        {previous ? 
          <a class="underline" href={`/escuela/cursos/${courseSlug}/${previous.slug}`}>← {previous.title}</a> 
          : <span></span>}
        {next ? 
          <a class="underline" href={`/escuela/cursos/${courseSlug}/${next.slug}`}>{next.title} →</a> 
          : <a class="underline" href="/escuela/mi-escuela">Volver a mi curso</a>}
      </nav>
  </main>

  </script>

  </script>
</Layout>
