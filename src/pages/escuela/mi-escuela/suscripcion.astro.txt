---
/**
 * Esta página permite a los clientes consultar y administrar su suscripción
 */

export const prerender = false;

import { env } from 'cloudflare:workers';
import Layout from '../../../layouts/Layout.astro';
import {
  accountsEnabled,
  getCustomerEmail,
} from '../../../features/auth/customer';

// La página requiere una cuenta de cliente.
if (!(await accountsEnabled())) {
  return Astro.redirect('/', 303);
}

const customerEmail = await getCustomerEmail(Astro.cookies);

if (!customerEmail) {
  return Astro.redirect('/account/login', 303);
}

// TODO: Consultar la suscripción del cliente utilizando
// las tablas y los helpers reales del proyecto.
//
// No se debe asumir que una suscripción está activa
// solamente porque exista un registro en D1.
//
// Tampoco se debe cancelar una suscripción modificando
// únicamente la base de datos: la operación debe
// sincronizarse con el proveedor de pagos.

const subscription = null;

const statusLabels: Record<string, string> = {
  active: 'Activa',
  trialing: 'En periodo de prueba',
  past_due: 'Pago pendiente',
  unpaid: 'Pago pendiente',
  canceled: 'Cancelada',
  incomplete: 'Pendiente de activación',
  incomplete_expired: 'Activación vencida',
  paused: 'Pausada',
};

const status = subscription?.status ?? null;
const statusLabel = status
  ? statusLabels[status] ?? status
  : 'Sin suscripción registrada';

const hasSubscription = Boolean(subscription);
---

        Consulta el estado de tu suscripción y administra tus opciones de pago

        <div>
          <p class="text-sm text-kumo-muted">Tu suscripción</p>
          <h2 class="mt-1 text-xl font-semibold">

          {statusLabel}

      </div>

      <div class="my-6 border-t border-kumo-line"></div>

      {hasSubscription ? (
        <div class="space-y-5">
          <div>
            <h3 class="font-medium">Estado de la renovación</h3>
            <p class="mt-1 text-sm leading-6 text-kumo-muted">

              Utiliza el portal seguro de tu proveedor de pagos para modificar

              renovación, cuando esa integración esté habilitada.
            </p>
          </div>
        </div>
      ) : (
        <div>
          <h3 class="font-semibold">No encontramos una suscripción</h3>
          <p class="mt-2 text-sm leading-6 text-kumo-muted">
            Si ya realizaste un pago, verifica que hayas ingresado con el
            correo electrónico utilizado al contratar la suscripción.
          </p>
          <a
            href="/products/suscripcion-escuela-matematicas"
            class="mt-5 inline-flex items-center justify-center rounded-lg bg-kumo-inverted px-5 py-3 text-sm font-semibold text-kumo-inverted-foreground"
          >
            Ver suscripción
          </a>
        </div>
      )}
    </section>

    <section class="mt-6 rounded-xl border border-kumo-line p-5 sm:p-6">
      <h2 class="text-lg font-semibold">¿Necesitas ayuda?</h2>
      <p class="mt-2 text-sm leading-6 text-kumo-muted">
        Si tienes un problema con un cobro o necesitas ayuda con tu cuenta,

      >
        Ir a mi cuenta
      </a>
    </section>
