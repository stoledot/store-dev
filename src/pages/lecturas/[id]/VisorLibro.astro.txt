export type NodoXML =
  | { tipo: "texto"; valor: string }
  | {
      tipo: "elemento";
      etiqueta: string;
      atributos: Record<string, string>;
      hijos: NodoXML[];
    };

  autor?: string;
  contenidoHtml?: string;
  contenidoXml?: NodoXML | NodoXML[];
}

function renderizarNodoXML(nodo: NodoXML | NodoXML[]): string {
  if (Array.isArray(nodo)) {
    return nodo.map(renderizarNodoXML).join("");
  }

  if (nodo.tipo === "texto") {
    return nodo.valor
      .replace(/&/g, "&amp;")
      .replace(/</g, "&lt;")
      .replace(/>/g, "&gt;");
  }

  const atributosStr = Object.entries(nodo.atributos || {})
    .map(([k, v]) => `${k}="${v.replace(/"/g, "&quot;")}"`)
    .join(" ");

  const hijosHtml = (nodo.hijos || []).map(renderizarNodoXML).join("");
  const apertura = atributosStr ? `<${nodo.etiqueta} ${atributosStr}>` : `<${nodo.etiqueta}>`;

  return `${apertura}${hijosHtml}</${nodo.etiqueta}>`;
}

    async function renderizarMathJaxEnFuente(contenedor: HTMLElement) {

        // Si MathJax v3 está cargando, esperamos su promesa startup

        if (typeof mj.typesetPromise === "function") {

          mj.typeset([contenedor]);
        }
      } catch (err) {
        console.warn("Error al renderizar MathJax antes de paginar:", err);
      }
    }

    function dividirParrafoConBusquedaBinaria(
        parrafo: HTMLElement,
        medidorEl: HTMLElement,
        paginasHTML: string[]
      ): HTMLElement | null 
    {
      const palabras = (parrafo.textContent || "").split(/\s+/).filter(Boolean);
      if (palabras.length === 0) return null;

      const pActual = parrafo.cloneNode(false) as HTMLElement;
      medidorEl.appendChild(pActual);

      let inicio = 1;
      let fin = palabras.length;
      let mejorCorte = 0;

      while (inicio <= fin) {
        const medio = Math.floor((inicio + fin) / 2);
        pActual.textContent = palabras.slice(0, medio).join(" ");

        if (medidorEl.scrollHeight <= medidorEl.clientHeight) {
          mejorCorte = medio;
          inicio = medio + 1;
        } else {
          fin = medio - 1;
        }
      }

      if (mejorCorte === 0) {
        if (medidorEl.children.length > 1) {
          medidorEl.removeChild(pActual);
          paginasHTML.push(medidorEl.innerHTML);
          medidorEl.innerHTML = "";
          return parrafo;
        }
        pActual.textContent = palabras[0];
        mejorCorte = 1;
      } else {
        pActual.textContent = palabras.slice(0, mejorCorte).join(" ");
      }

      if (mejorCorte >= palabras.length) {
        return null;
      }

      paginasHTML.push(medidorEl.innerHTML);
      medidorEl.innerHTML = "";

      const pRestante = parrafo.cloneNode(false) as HTMLElement;
      pRestante.style.textIndent = "0";
      pRestante.textContent = palabras.slice(mejorCorte).join(" ");
      return pRestante;
    }

    function construirLibro(paginaInicial = 0) {
      const mesaLectura = document.querySelector(".mesa-lectura");
      if (!mesaLectura) return;

      // 1. Si ya existía una instancia, la destruimos y recreamos el contenedor limpio
      if (pageFlip) {
        pageFlip.destroy();
        pageFlip = null;
      }

      // Recrear #libro-contenedor porque pageFlip.destroy() lo remueve del DOM
      mesaLectura.innerHTML = `<div id="libro-contenedor" class="libro-wrapper"></div>`;
      const contenedorActual = document.getElementById("libro-contenedor")!;

      if (pageFlip) {
        pageFlip.destroy();
        pageFlip = null;
      }
      contenedorLibro!.innerHTML = "";

      const { anchoHoja, altoHoja, esMovil } = calcularDimensionesHoja();
      sincronizarMedidorConHojaReal(anchoHoja, altoHoja);

      const colaNodos: HTMLElement[] = obtenerBloquesPaginables(fuenteContenido!).map(
        (n) => n.cloneNode(true) as HTMLElement
      );

      const paginasHTML: string[] = [];
      medidor!.innerHTML = "";

      const MAX_ITERACIONES = 5000;
      let iteraciones = 0;

      while (colaNodos.length > 0 && iteraciones < MAX_ITERACIONES) {
        iteraciones++;
        const nodoActual = colaNodos.shift()!;

        medidor!.appendChild(nodoActual);

        if (medidor!.scrollHeight <= medidor!.clientHeight) {
          continue;
        }

        medidor!.removeChild(nodoActual);

        if (nodoActual.tagName === "P" && !nodoActual.querySelector("svg, img, table, mjx-container, .MathJax")) {
          const sobrante = dividirParrafoConBusquedaBinaria(nodoActual, medidor!, paginasHTML);
          if (sobrante) {
            colaNodos.unshift(sobrante);
          }
        } else {
          if (medidor!.children.length > 0) {
            paginasHTML.push(medidor!.innerHTML);
            medidor!.innerHTML = "";
            colaNodos.unshift(nodoActual);
          } else {
            medidor!.appendChild(nodoActual);
            paginasHTML.push(medidor!.innerHTML);
            medidor!.innerHTML = "";
          }
        }
      }

      if (medidor!.innerHTML.trim() !== "") {
        paginasHTML.push(medidor!.innerHTML);
      }
      medidor!.innerHTML = "";

      const fragmento = document.createDocumentFragment();

      const portada = document.createElement("div");
      portada.className = "pagina pagina-dura portada";
      portada.setAttribute("data-density", "hard");
      portada.innerHTML = `
        <div class="contenido-portada">
          <div class="marco-dorado">
            <p class="etiqueta-editorial">EDICIÓN DIGITAL</p>
            <h1>${titulo}</h1>
            <div class="ornamento">❧</div>
            <p class="autor-portada">${autor}</p>
          </div>
        </div>
      `;
      fragmento.appendChild(portada);

      if (!esMovil) {
        const guardaInicio = document.createElement("div");
        guardaInicio.className = "pagina pagina-guarda";
        guardaInicio.innerHTML = `<div class="contenido-pagina"></div>`;
        fragmento.appendChild(guardaInicio);
      }

      paginasHTML.forEach((html, idx) => {
        const numPagina = idx + 1;
        const esPar = numPagina % 2 === 0;
        const pagina = document.createElement("div");
        pagina.className = `pagina pagina-papel ${esPar ? "pagina-izquierda" : "pagina-derecha"}`;
        pagina.innerHTML = `
          <div class="contenido-pagina">
            <div class="encabezado-pagina">${esPar ? titulo : autor}</div>
            <div class="cuerpo-texto">${html}</div>
            <div class="pie-pagina">${numPagina}</div>
          </div>
          <div class="sombra-lomo"></div>
        `;
        fragmento.appendChild(pagina);
      });

      if (!esMovil && paginasHTML.length % 2 !== 0) {
        const paginaExtra = document.createElement("div");
        paginaExtra.className = "pagina pagina-papel pagina-izquierda";
        paginaExtra.innerHTML = `<div class="contenido-pagina"></div><div class="sombra-lomo"></div>`;
        fragmento.appendChild(paginaExtra);
      }

      const contraportada = document.createElement("div");
      contraportada.className = "pagina pagina-dura contraportada";
      contraportada.setAttribute("data-density", "hard");
      contraportada.innerHTML = `
        <div class="contenido-portada">
          <div class="ornamento">❦</div>
          <p class="fin-texto">FIN</p>
        </div>
      `;
      fragmento.appendChild(contraportada);

            // 1. Donde agregas las páginas:
      contenedorActual.appendChild(fragmento);

      // 2. Donde inicializas PageFlip:
      pageFlip = new PageFlip(contenedorActual, {
        width: anchoHoja,
        height: altoHoja,
        size: "fixed",
        maxShadowOpacity: 0.45,
        showCover: true,
        mobileScrollSupport: false,
        usePortrait: esMovil,
        flippingTime: 750,
        startPage: Math.min(paginaInicial, paginasHTML.length + 1),
      });


      pageFlip.loadFromHTML(document.querySelectorAll(".pagina"));

      const totalPaginas = pageFlip.getPageCount();

      const actualizarEstadoUI = (indiceActual: number) => {
        if (!indicadorPagina) return;
        if (indiceActual === 0) {
          indicadorPagina.textContent = "Portada";
        } else if (indiceActual >= totalPaginas - 1) {
          indicadorPagina.textContent = "Contraportada";
        } else {
          indicadorPagina.textContent = `Pág. ${indiceActual} de ${totalPaginas - 2}`;
        }
        if (btnAnterior) btnAnterior.disabled = indiceActual === 0;
        if (btnSiguiente) btnSiguiente.disabled = indiceActual >= totalPaginas - 1;
      };

      pageFlip.on("flip", (e) => {
        actualizarEstadoUI(e.data as number);
      });

      actualizarEstadoUI(pageFlip.getCurrentPageIndex());
    }

    btnAnterior?.addEventListener("click", () => pageFlip?.flipPrev());
    btnSiguiente?.addEventListener("click", () => pageFlip?.flipNext());

    document.addEventListener("keydown", (e) => {
      if (e.key === "ArrowLeft") pageFlip?.flipPrev();
      if (e.key === "ArrowRight") pageFlip?.flipNext();
    });

    let timeoutResize: number;
    window.addEventListener("resize", () => {
      if (window.innerWidth === anchoActual) return;
      anchoActual = window.innerWidth;

      clearTimeout(timeoutResize);
      timeoutResize = window.setTimeout(() => {
        const paginaActual = pageFlip ? pageFlip.getCurrentPageIndex() : 0;
        construirLibro(paginaActual);
      }, 250);
    });

        // Primero renderizamos MathJax en el contenido oculto y LUEGO calculamos las páginas
    (async () => {
      if (fuenteContenido) {
        await renderizarMathJaxEnFuente(fuenteContenido);
      }
      construirLibro(0);
    })();


    construirLibro(0);
  });
