<script>
  const fullscreenButton = document.querySelector("#fb-fullscreen");
  const readerLayout = document.querySelector("#fb-reader-layout");

  fullscreenButton?.addEventListener("click", async () => {
    if (!document.fullscreenElement) {
      await readerLayout?.requestFullscreen?.();
    } else {
      await document.exitFullscreen?.();
    }
  });

  const sidebar = document.querySelector("#fb-sidebar");
  const sidebarOverlay = document.querySelector("#fb-sidebar-overlay");
  const tocToggle = document.querySelector("#fb-toc-toggle");
  const sidebarClose = document.querySelector("#fb-sidebar-close");

  const openSidebar = () => {
    sidebar?.classList.remove("-translate-x-full");
    sidebarOverlay?.classList.remove("hidden");
  };

  const closeSidebar = () => {
    sidebar?.classList.add("-translate-x-full");
    sidebarOverlay?.classList.add("hidden");
  };

  tocToggle?.addEventListener("click", () => {
    if (sidebar?.classList.contains("-translate-x-full")) {
      openSidebar();
    } else {
      closeSidebar();
    }
  });

  sidebarClose?.addEventListener("click", closeSidebar);
  sidebarOverlay?.addEventListener("click", closeSidebar);

  // En pantallas grandes el índice siempre permanece visible.
  const handleResponsiveSidebar = () => {
    if (window.innerWidth >= 1024) {
      sidebar?.classList.remove("-translate-x-full");
      sidebarOverlay?.classList.add("hidden");
    }
  };

  window.addEventListener("resize", handleResponsiveSidebar);
</script>

<script>
  const searchInput = document.querySelector<HTMLInputElement>(
    "[data-reading-toc-search]"
  );

  const tocOriginal = document.querySelector("#reading-toc");
  const tocResults = document.querySelector("#reading-toc-results");

  searchInput?.addEventListener("input", () => {
    const query = searchInput.value.trim().toLowerCase();

    if (!query) {
      tocOriginal?.classList.remove("hidden");
      tocResults?.classList.add("hidden");
      if (tocResults) tocResults.innerHTML = "";
      return;
    }

    const links = tocOriginal?.querySelectorAll("a") ?? [];

    const matchingLinks = Array.from(links).filter((link) =>
      link.textContent?.toLowerCase().includes(query)
    );

    tocOriginal?.classList.add("hidden");
    tocResults?.classList.remove("hidden");

    if (!tocResults) return;

    if (matchingLinks.length === 0) {
      tocResults.innerHTML = `
        <div class="py-10 text-center text-sm text-slate-500">
          No se encontraron temas
        </div>
      `;
      return;
    }

    tocResults.innerHTML = "";

    matchingLinks.forEach((link) => {
      const clone = link.cloneNode(true) as HTMLAnchorElement;

      clone.classList.add(
        "block",
        "rounded-xl",
        "border",
        "border-white/5",
        "bg-white/5",
        "px-4",
        "py-3",
        "text-sm",
        "text-slate-300",
        "transition",
        "hover:bg-white/10",
        "hover:text-white"
      );

      const wrapper = document.createElement("div");
      wrapper.className = "mb-2";
      wrapper.appendChild(clone);

      tocResults.appendChild(wrapper);
    });
  });
</script>
